# توجيهات nginx الإضافية للساب دومين في Plesk
# المكان: Plesk › الساب دومين › Apache & nginx Settings › Additional nginx directives
# (مع تفعيل "Proxy mode" وشهادة Let's Encrypt من Plesk أولًا)

client_max_body_size 210m;

location / {
    proxy_pass http://127.0.0.1:8820;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_http_version 1.1;
    proxy_read_timeout 300s;
}

# بث SSE الحي (خريطة الشبكة) — بدون تخزين مؤقت
location /api/stream {
    proxy_pass http://127.0.0.1:8820;
    proxy_set_header Host $host;
    proxy_set_header X-Forwarded-For $proxy_add_x_forwarded_for;
    proxy_set_header X-Forwarded-Proto $scheme;
    proxy_http_version 1.1;
    proxy_buffering off;
    proxy_cache off;
    proxy_read_timeout 24h;
    chunked_transfer_encoding off;
}

# ملاحظة WAF (ModSecurity): لا ترسل POST بجسم فارغ للمنصة — أرسل '{}' على الأقل.
